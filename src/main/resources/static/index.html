⌨ Commands used in this project

# build the jar and the image
mvn clean package
docker build -t spring-boot-mysql-app .
# start both containers with one command
docker-compose up -d
docker-compose ps
# look inside the database
docker exec -it <db-container-name> /bin/bash
mysql -u root -p      # password: root
show databases;  use sbms;  show tables;  select * from book;
# stop and remove everything
docker-compose down
